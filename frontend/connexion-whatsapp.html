// ✅ VERSION FINALE DE verifierCode() function verifierCode() { let inputs = document.querySelectorAll('.code-box'), code = ''; inputs.forEach(i => code += i.value); if (code.length < 6) return showError('Code incomplet'); fetch('/api/whatsapp/verify-code', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ telephone: currentPhone, code }) }) .then(res => res.json()) .then(data => { console.log("📦 Données reçues :", data); console.log("🎯 Rôle reçu :", data.role); if (data.success) { if (data.token) { localStorage.setItem('token', data.token); console.log("✅ Token stocké"); } if (data.user) { localStorage.setItem('user', JSON.stringify(data.user)); console.log("✅ Utilisateur stocké"); } updateSteps(3); // ✅ LOGIQUE DE REDIRECTION OPTIMISÉE setTimeout(() => { let url = 'index.html'; // URL par défaut // 1. Gérer en priorité l'inscription requise if (data.require_inscription) { url = 'inscription-patient.html?tel=' + encodeURIComponent(currentPhone); } // 2. Si l'utilisateur est déjà inscrit, vérifier son rôle else if (data.role) { if (data.role === 'patient') url = 'dashboard-patient.html'; else if (data.role === 'medecin') url = 'dashboard-medecin.html'; else if (data.role === 'infirmier') url = 'dashboard-infirmier.html'; else if (data.role === 'administratif') url = 'dashboard-admin.html'; else if (data.role === 'hotellerie') url = 'hotellerie.html'; else if (data.role === 'logistique') url = 'stocks.html'; else if (data.role === 'qualite') url = 'qualite.html'; else if (data.role === 'voyages' || data.role === 'chauffeur') url = 'voyages.html'; else if (data.role === 'direction' || data.role === 'boss') url = 'dashboard-direction.html'; } // 3. Cas où rien n'est défini (erreur ou visiteur) else { console.warn("⚠️ Aucun rôle ni inscription requise trouvés, redirection par défaut."); } console.log("🔀 Redirection vers :", url); window.location.href = url; }, 2000); } else { showError(data.error || 'Code incorrect'); inputs.forEach(i => i.value = ''); inputs[0].focus(); } }) .catch(err => { console.error("❌ Erreur réseau :", err); showError('Erreur serveur'); }); }
